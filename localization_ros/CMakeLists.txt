cmake_minimum_required(VERSION 3.16)
project(mgts_localization)
find_package(ament_cmake REQUIRED)
find_package(Sophus REQUIRED)
find_package(beluga REQUIRED)
find_package(beluga_vdb REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(beluga_node src/beluga_node.cpp)
target_compile_features(beluga_node PRIVATE cxx_std_17)
target_compile_options(beluga_node PRIVATE -O2)
target_link_libraries(beluga_node beluga::beluga beluga_vdb::beluga_vdb)
ament_target_dependencies(beluga_node rclcpp sensor_msgs nav_msgs geometry_msgs diagnostic_msgs std_srvs)
add_executable(prepare_vdb src/prepare_vdb.cpp)
target_compile_features(prepare_vdb PRIVATE cxx_std_17)
target_link_libraries(prepare_vdb PRIVATE beluga_vdb::beluga_vdb)
install(TARGETS beluga_node prepare_vdb DESTINATION lib/${PROJECT_NAME})
ament_package()
