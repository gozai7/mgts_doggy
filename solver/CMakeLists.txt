cmake_minimum_required(VERSION 3.16)
project(mgts_graph)
find_package(ament_cmake REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(TBB REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization)
find_path(KARTO_INCLUDE karto_sdk/Mapper.h REQUIRED)
find_library(KARTO_LIBRARY kartoSlamToolbox REQUIRED)
add_executable(export_graph export_graph.cpp)
ament_target_dependencies(export_graph rclcpp_lifecycle)
target_include_directories(export_graph PRIVATE ${KARTO_INCLUDE})
target_link_libraries(export_graph Eigen3::Eigen TBB::tbb Boost::serialization ${KARTO_LIBRARY})
install(TARGETS export_graph DESTINATION bin)

if(NOT KARTO_SOURCE)
  return()
endif()

add_library(directional_karto SHARED ${KARTO_SOURCE}/Mapper.cpp ${KARTO_SOURCE}/Karto.cpp)
target_compile_features(directional_karto PUBLIC cxx_std_17)
ament_target_dependencies(directional_karto rclcpp_lifecycle)
target_include_directories(directional_karto PUBLIC ${KARTO_INCLUDE} PRIVATE karto)
find_package(Boost REQUIRED COMPONENTS serialization filesystem system thread)
target_link_libraries(directional_karto Eigen3::Eigen TBB::tbb ${Boost_LIBRARIES})
set_target_properties(directional_karto PROPERTIES OUTPUT_NAME kartoSlamToolbox)
install(TARGETS directional_karto DESTINATION lib/mgts-directional)

add_executable(test_directional karto/test_directional.cpp)
target_link_libraries(test_directional directional_karto)
enable_testing()
add_test(NAME directional COMMAND test_directional)
